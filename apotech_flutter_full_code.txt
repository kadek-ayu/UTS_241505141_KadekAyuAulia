class Medicine {
  final int? id;
  final String name;
  final String category;
  final String description;
  final String packaging;
  final int stock;
  final DateTime? createdAt;

  const Medicine({
    this.id,
    required this.name,
    required this.category,
    required this.description,
    required this.packaging,
    required this.stock,
    this.createdAt,
  });

  factory Medicine.fromJson(Map<String, dynamic> json) {
    return Medicine(
      id: json['id'] as int?,
      name: json['name'] as String,
      category: json['category'] as String,
      description: json['description'] as String,
      packaging: json['packaging'] as String,
      stock: json['stock'] as int,
      createdAt: json['created_at'] == null
          ? null
          : DateTime.parse(json['created_at'] as String),
    );
  }

  Map<String, dynamic> toJson() {
    return {
      'name': name,
      'category': category,
      'description': description,
      'packaging': packaging,
      'stock': stock,
    };
  }
}

class OpenFdaProduct {
  final String productNdc;
  final String genericName;
  final String brandName;
  final String manufacturer;
  final String dosageForm;
  final String productType;
  final List<String> routes;
  final List<String> activeIngredients;

  const OpenFdaProduct({
    required this.productNdc,
    required this.genericName,
    required this.brandName,
    required this.manufacturer,
    required this.dosageForm,
    required this.productType,
    required this.routes,
    required this.activeIngredients,
  });

  factory OpenFdaProduct.fromJson(Map<String, dynamic> json) {
    final ingredients = json['active_ingredients'];

    return OpenFdaProduct(
      productNdc: json['product_ndc'] ?? '',
      genericName: json['generic_name'] ?? '',
      brandName: json['brand_name'] ?? '',
      manufacturer: json['labeler_name'] ?? '',
      dosageForm: json['dosage_form'] ?? '',
      productType: json['product_type'] ?? '',
      routes: json['route'] is List
          ? List<String>.from(json['route'])
          : [],
      activeIngredients: ingredients is List
          ? ingredients
              .whereType<Map<String, dynamic>>()
              .map((item) => item['name']?.toString() ?? '')
              .where((name) => name.isNotEmpty)
              .toList()
          : [],
    );
  }
}

import 'package:flutter/foundation.dart';

class CartItem {
  final int pharmacyId;
  final String pharmacyName;
  final int medicineId;
  final String name;
  final double price;
  final int stockQuantity;
  int quantity;

  CartItem({
    required this.pharmacyId,
    required this.pharmacyName,
    required this.medicineId,
    required this.name,
    required this.price,
    required this.stockQuantity,
    this.quantity = 1,
  });

  double get subtotal => price * quantity;
}

enum CartAddResult {
  added,
  differentPharmacy,
  outOfStock,
  insufficientStock,
}

class CartProvider extends ChangeNotifier {
  final List<CartItem> _items = [];

  int? _pharmacyId;
  String? _pharmacyName;

  List<CartItem> get items => List.unmodifiable(_items);

  int? get pharmacyId => _pharmacyId;

  String? get pharmacyName => _pharmacyName;

  bool get isEmpty => _items.isEmpty;

  int get itemCount {
    return _items.fold(
      0,
      (total, item) => total + item.quantity,
    );
  }

  double get totalPrice {
    return _items.fold(
      0,
      (total, item) => total + item.subtotal,
    );
  }

  CartAddResult addItem({
    required int pharmacyId,
    required String pharmacyName,
    required int medicineId,
    required String name,
    required double price,
    required int stockQuantity,
  }) {
    if (stockQuantity <= 0) {
      return CartAddResult.outOfStock;
    }

    if (_items.isNotEmpty && _pharmacyId != pharmacyId) {
      return CartAddResult.differentPharmacy;
    }

    final existingIndex = _items.indexWhere(
      (item) => item.medicineId == medicineId,
    );

    if (existingIndex != -1) {
      final existingItem = _items[existingIndex];

      if (existingItem.quantity >= existingItem.stockQuantity) {
        return CartAddResult.insufficientStock;
      }

      existingItem.quantity++;
    } else {
      _pharmacyId = pharmacyId;
      _pharmacyName = pharmacyName;

      _items.add(
        CartItem(
          pharmacyId: pharmacyId,
          pharmacyName: pharmacyName,
          medicineId: medicineId,
          name: name,
          price: price,
          stockQuantity: stockQuantity,
        ),
      );
    }

    notifyListeners();
    return CartAddResult.added;
  }

  bool increaseQuantity(int medicineId) {
    final index = _items.indexWhere(
      (item) => item.medicineId == medicineId,
    );

    if (index == -1) return false;

    final item = _items[index];

    if (item.quantity >= item.stockQuantity) {
      return false;
    }

    item.quantity++;
    notifyListeners();
    return true;
  }

  bool decreaseQuantity(int medicineId) {
    final index = _items.indexWhere(
      (item) => item.medicineId == medicineId,
    );

    if (index == -1) return false;

    final item = _items[index];

    if (item.quantity > 1) {
      item.quantity--;
      notifyListeners();
      return true;
    }

    return false;
  }

  void removeItem(int medicineId) {
    _items.removeWhere(
      (item) => item.medicineId == medicineId,
    );

    _resetPharmacyIfEmpty();
    notifyListeners();
  }

  void clearCart() {
    _items.clear();
    _pharmacyId = null;
    _pharmacyName = null;
    notifyListeners();
  }

  void _resetPharmacyIfEmpty() {
    if (_items.isEmpty) {
      _pharmacyId = null;
      _pharmacyName = null;
    }
  }
}

import 'package:flutter/material.dart';

import '../models/openfda_product.dart';

class FavoriteProvider extends ChangeNotifier {
  final Map<String, OpenFdaProduct> _favorites = {};

  List<OpenFdaProduct> get favorites =>
      _favorites.values.toList();

  bool isFavorite(OpenFdaProduct product) {
    return _favorites.containsKey(product.productNdc);
  }

  void toggleFavorite(OpenFdaProduct product) {
    final productId = product.productNdc;

    if (productId.isEmpty) {
      return;
    }

    if (_favorites.containsKey(productId)) {
      _favorites.remove(productId);
    } else {
      _favorites[productId] = product;
    }

    notifyListeners();
  }

  void removeFavorite(OpenFdaProduct product) {
    _favorites.remove(product.productNdc);
    notifyListeners();
  }

  void clearFavorites() {
    _favorites.clear();
    notifyListeners();
  }
}

  bool _isLoading = false;
  String? _errorMessage;

  List<Medicine> get medicines => _medicines;

    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      _medicines = await _repository.getMedicines();
    } catch (error) {
      _errorMessage = 'Gagal memuat data produk.';
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }
}

import 'package:flutter/foundation.dart';

import '../services/order_service.dart';
import 'cart_provider.dart';

class OrderProvider extends ChangeNotifier {
  final OrderService _orderService = OrderService();

  bool _isLoading = false;
  String? _errorMessage;
  Map<String, dynamic>? _lastOrder;
  List<Map<String, dynamic>> _orders = [];

  bool get isLoading => _isLoading;
  String? get errorMessage => _errorMessage;
  Map<String, dynamic>? get lastOrder => _lastOrder;
  List<Map<String, dynamic>> get orders =>
      List.unmodifiable(_orders);

  Future<bool> createOrder(CartProvider cart) async {
    if (_isLoading) return false;

    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      _lastOrder = await _orderService.createOrder(
        cart: cart,
      );

      return true;
    } catch (error) {
      _errorMessage = error.toString();
      return false;
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  Future<void> loadOrders() async {
    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      _orders = await _orderService.getOrders();
    } catch (error) {
      _errorMessage = error.toString();
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }
}

import '../../providers/cart_provider.dart';
import '../../providers/order_provider.dart';

class CheckoutScreen extends StatelessWidget {
  const CheckoutScreen({super.key});

  String _formatPrice(double price) {
    return 'Rp ${price.toStringAsFixed(0)}';
  }

  Future<void> _confirmOrder(BuildContext context) async {
    final confirmed = await showDialog<bool>(
      context: context,
      builder: (dialogContext) {
        return AlertDialog(
          title: const Text('Konfirmasi Pesanan'),
          content: const Text(
            'Apakah Anda yakin ingin mengonfirmasi pesanan ini? '
            'Pesanan ini merupakan simulasi dan perlu dikonfirmasi '
            'dalam waktu 24 jam.',
          ),
          actions: [
            TextButton(
              onPressed: () {
                Navigator.pop(dialogContext, false);
              },
              child: const Text('Periksa Kembali'),
            ),
            ElevatedButton(
              onPressed: () {
                Navigator.pop(dialogContext, true);
              },
              child: const Text('Ya, Konfirmasi'),
            ),
          ],
        );
      },
    );

    if (confirmed != true || !context.mounted) return;

    final cart = context.read<CartProvider>();
    final orderProvider = context.read<OrderProvider>();

    final success = await orderProvider.createOrder(cart);

    if (!context.mounted) return;

    if (success) {
      final order = orderProvider.lastOrder;

      final orderId = order?['id']?.toString() ?? '-';

      final receiptNumber = orderId == '-'
          ? '-'
          : 'APT-${orderId.padLeft(6, '0')}';

      final totalPrice =
          (order?['total_price'] as num?)?.toDouble() ?? 0;

      await showDialog<void>(
        context: context,
        barrierDismissible: false,
        builder: (dialogContext) {
          return AlertDialog(
            icon: const Icon(
              Icons.check_circle_outline,
              color: Colors.green,
              size: 55,
            ),
            title: const Text('Pesanan Berhasil!'),
            content: Column(
              mainAxisSize: MainAxisSize.min,
              crossAxisAlignment: CrossAxisAlignment.start,
              children: [
                const Text(
                  'Pesanan Anda berhasil dibuat dan tersimpan.',
                ),
                const SizedBox(height: 20),
                const Text(
                  'Nomor Resi',
                  style: TextStyle(
                    color: Colors.black54,
                    fontSize: 12,
                  ),
                ),
                const SizedBox(height: 4),
                SelectableText(
                  receiptNumber,
                  style: const TextStyle(
                    fontWeight: FontWeight.bold,
                    fontSize: 18,
                    color: Colors.green,
                  ),
                ),
                const SizedBox(height: 16),
                const Text(
                  'Total Pesanan',
                  style: TextStyle(
                    color: Colors.black54,
                    fontSize: 12,
                  ),
                ),
                const SizedBox(height: 4),
                Text(
                  _formatPrice(totalPrice),
                  style: const TextStyle(
                    fontWeight: FontWeight.bold,
                    fontSize: 17,
                  ),
                ),
                const SizedBox(height: 16),
                const Text(
                  'Catatan: Pesanan ini merupakan simulasi. '
                  'Tidak ada transaksi pembayaran atau '
                  'pengambilan sungguhan.',
                  style: TextStyle(fontSize: 12),
                ),
              ],
            ),
            actions: [
              SizedBox(
                width: double.infinity,
                child: ElevatedButton(
                  onPressed: () {
                    Navigator.pop(dialogContext);
                    Navigator.pop(context);
                  },
                  child: const Text('Baik'),
                ),
              ),
            ],
          );
        },
      );
    } else {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text(
            orderProvider.errorMessage ??
                'Pesanan gagal dibuat.',
          ),
          backgroundColor: Colors.red,
        ),
      );
    }
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(
        title: const Text('Konfirmasi Pesanan'),
      ),
      body: Consumer<CartProvider>(
        builder: (context, cart, child) {
          if (cart.isEmpty) {
            return const Center(
              child: Text('Keranjang masih kosong.'),
            );
          }

          return ListView(
            padding: const EdgeInsets.all(16),
            children: [
              const Text(
                'Informasi Apotek',
                style: TextStyle(
                  fontSize: 17,
                  fontWeight: FontWeight.bold,
                ),
              ),
              const SizedBox(height: 10),
              Card(
                child: ListTile(
                  leading: const Icon(
                    Icons.local_pharmacy,
                    color: Colors.green,
                  ),
                  title: Text(cart.pharmacyName ?? '-'),
                  subtitle: const Text('Apotek pilihan Anda'),
                ),
              ),
              const SizedBox(height: 24),
              const Text(
                'Ringkasan Produk',
                style: TextStyle(
                  fontSize: 17,
                  fontWeight: FontWeight.bold,
                ),
              ),
              const SizedBox(height: 10),
              ...cart.items.map(
                (item) => Card(
                  child: ListTile(
                    title: Text(item.name),
                    subtitle: Text(
                      '${item.quantity} x '
                      '${_formatPrice(item.price)}',
                    ),
                    trailing: Text(
                      _formatPrice(item.subtotal),
                      style: const TextStyle(
                        fontWeight: FontWeight.bold,
                      ),
                    ),
                  ),
                ),
              ),
              const SizedBox(height: 20),
              Card(
                color: Colors.green.shade50,
                child: Padding(
                  padding: const EdgeInsets.all(16),
                  child: Column(
                    children: [
                      const Text(
                        'Total Pesanan',
                        style: TextStyle(fontSize: 14),
                      ),
                      const SizedBox(height: 8),
                      Text(
                        _formatPrice(cart.totalPrice),
                        style: const TextStyle(
                          fontSize: 23,
                          fontWeight: FontWeight.bold,
                          color: Colors.green,
                        ),
                      ),
                    ],
                  ),
                ),
              ),
              const SizedBox(height: 16),
              Container(
                padding: const EdgeInsets.all(14),
                decoration: BoxDecoration(
                  color: Colors.orange.shade50,
                  borderRadius: BorderRadius.circular(12),
                ),
                child: const Text(
                  'Catatan: Ini merupakan pemesanan simulasi. '
                  'Pesanan perlu dikonfirmasi dalam waktu 24 jam. '
                  'Belum ada transaksi atau pengambilan sungguhan.',
                  style: TextStyle(fontSize: 12),
                ),
              ),
              const SizedBox(height: 24),
              Consumer<OrderProvider>(
                builder: (context, orderProvider, child) {
                  return ElevatedButton(
                    onPressed: orderProvider.isLoading
                        ? null
                        : () => _confirmOrder(context),
                    style: ElevatedButton.styleFrom(
                      minimumSize: const Size(
                        double.infinity,
                        50,
                      ),
                    ),
                    child: orderProvider.isLoading
                        ? const CircularProgressIndicator()
                        : const Text('Konfirmasi Pesanan'),
                  );
                },
              ),
            ],
          );
        },
      ),
    );
  }
}

          ),
        ),
      ),
      body: Consumer<FavoriteProvider>(
        builder: (context, favoriteProvider, child) {
          final favorites = favoriteProvider.favorites;

          if (favorites.isEmpty) {
            return const Center(
              child: Column(
                mainAxisSize: MainAxisSize.min,
                children: [
                  Icon(
                    Icons.star_border,
                    size: 64,
                    color: Colors.grey,

                    'untuk menyimpannya.',
                    textAlign: TextAlign.center,
                    style: TextStyle(
                      color: Colors.grey,
                      fontSize: 13,
                    ),
                  ),
                ],
              ),
            );
          }

          return ListView.separated(
            padding: const EdgeInsets.all(16),
            itemCount: favorites.length,
            separatorBuilder: (context, index) =>
                const SizedBox(height: 12),
            itemBuilder: (context, index) {
              final product = favorites[index];

              final displayName = product.brandName.isNotEmpty
                  ? product.brandName
                  : product.genericName;

              return Card(
                elevation: 1,
                color: Colors.white,
                shape: RoundedRectangleBorder(
                  borderRadius: BorderRadius.circular(14),
                ),
                clipBehavior: Clip.antiAlias,
                child: ListTile(

                      borderRadius: BorderRadius.circular(12),
                    ),
                    child: const Icon(
                      Icons.medication_outlined,
                      color: AppTheme.primaryColor,
                    ),
                  ),
                  title: Text(
                    displayName.isNotEmpty
                        ? displayName
                        : 'Nama produk tidak tersedia',
                    style: const TextStyle(
                      fontWeight: FontWeight.bold,
                    ),
                  ),
                  subtitle: Text(
                    product.genericName.isNotEmpty

                    overflow: TextOverflow.ellipsis,

                    tooltip: 'Hapus dari tersimpan',
                    icon: Icon(

                      color: Colors.amber.shade700,

                      favoriteProvider.toggleFavorite(product);

                      ScaffoldMessenger.of(context)
                        ..hideCurrentSnackBar()
                        ..showSnackBar(

                            content: Text(
                              'Produk dihapus dari tersimpan.',
                            ),
                          ),
                        );
                    },
                  ),
                  onTap: () {
                    Navigator.push(
                      context,
                      MaterialPageRoute(
                        builder: (context) =>
                            OpenFdaProductDetailScreen(
                          product: product,
                        ),
                      ),
                    );
                  },
                ),
              );
            },
          );
        },
      ),
    );

    final value = double.tryParse(price.toString()) ?? 0;

    return 'Rp ${value.toStringAsFixed(0).replaceAllMapped(
          RegExp(r'(\d)(?=(\d{3})+(?!\d))'),
          (match) => '${match[1]}.',
        )}';
  }

  String _formatDate(dynamic date) {
    final parsedDate = DateTime.tryParse(date.toString());

    if (parsedDate == null) return '-';

    final localDate = parsedDate.toLocal();

    const months = [
      'Januari',
      'Februari',
      'Maret',
      'April',
      'Mei',
      'Juni',
      'Juli',
      'Agustus',
      'September',
      'Oktober',
      'November',
      'Desember',
    ];

    return '${localDate.day} ${months[localDate.month - 1]} '
        '${localDate.year}, '
        '${localDate.hour.toString().padLeft(2, '0')}:'
        '${localDate.minute.toString().padLeft(2, '0')}';
  }

  Map<String, dynamic> _asMap(dynamic value) {
    if (value is Map) {
      return Map<String, dynamic>.from(value);
    }

    return {};
  }

    if (value is List) {
      return value
          .whereType<Map>()
          .map((item) => Map<String, dynamic>.from(item))

    }

    return [];

        return 'Selesai';
      case 'cancelled':
        return 'Dibatalkan';
      default:
        return status;
    }

    });

    try {
      final response = await supabase
          .from('pharmacies')
          .select()
          .eq('is_active', true)
          .order('name');

          .map((item) => Map<String, dynamic>.from(item))
          .toList();

        pharmacies = data;
        isLoading = false;
      });

      if (!mounted) return;

      setState(() {
        errorMessage = error.toString();
        isLoading = false;
      });
    }

import 'package:supabase_flutter/supabase_flutter.dart';

import '../models/openfda_product.dart';

class OpenFdaInventoryService {
  final SupabaseClient _supabase = Supabase.instance.client;

  Future<bool> saveProduct(OpenFdaProduct product) async {
    if (product.productNdc.isEmpty) {
      throw Exception('Kode produk NDC tidak tersedia.');
    }

    final existing = await _supabase
        .from('medicines')
        .select('id')
        .eq('product_ndc', product.productNdc)
        .maybeSingle();

    if (existing != null) {
      return false;
    }

    final description = [
      'Nama generik: ${product.genericName}',
      'Produsen: ${product.manufacturer}',
      'Bahan aktif: ${product.activeIngredients.join(', ')}',
    ].join('\n');

    await _supabase.from('medicines').insert({
      'product_ndc': product.productNdc,
      'name': product.brandName.isNotEmpty
          ? product.brandName
          : product.genericName,
      'category': product.productType,
      'description': description,
      'packaging': product.dosageForm,
      'stock': 0,
    });

    return true;
  }
}

import 'dart:convert';

import 'package:http/http.dart' as http;

import '../models/openfda_product.dart';

class OpenFdaService {
  static const String _baseUrl =
      'https://api.fda.gov/drug/ndc.json';

  Future<List<OpenFdaProduct>> getProducts() async {
    final uri = Uri.parse(_baseUrl).replace(
      queryParameters: {
        'search': 'product_type:"HUMAN OTC DRUG"',
        'limit': '20',
      },
    );

    final response = await http.get(uri);

    if (response.statusCode == 200) {
      final Map<String, dynamic> data =
          jsonDecode(response.body) as Map<String, dynamic>;

      final List<dynamic> results =
          data['results'] as List<dynamic>;

      return results
          .map(
            (item) => OpenFdaProduct.fromJson(
              item as Map<String, dynamic>,
            ),
          )
          .toList();
    }

    throw Exception(
      'Gagal mengambil data OpenFDA. '
      'Status code: ${response.statusCode}',
    );
  }
}

import 'package:supabase_flutter/supabase_flutter.dart';

import '../providers/cart_provider.dart';

class OrderService {
  final SupabaseClient _supabase = Supabase.instance.client;

  Future<Map<String, dynamic>> createOrder({
    required CartProvider cart,
  }) async {
    if (cart.isEmpty || cart.pharmacyId == null) {
      throw Exception('Keranjang masih kosong.');
    }

    final order = await _supabase
        .from('orders')
        .insert({
          'pharmacy_id': cart.pharmacyId,
          'total_price': cart.totalPrice,
          'order_status': 'pending',
        })
        .select()
        .single();

    final orderId = order['id'];

    try {
      final orderItems = cart.items.map((item) {
        return {
          'order_id': orderId,
          'medicine_id': item.medicineId,
          'quantity': item.quantity,
          'unit_price': item.price,
          'subtotal': item.subtotal,
        };
      }).toList();

      await _supabase
          .from('order_items')
          .insert(orderItems);

      return order;
    } catch (error) {
      await _supabase
          .from('orders')
          .delete()
          .eq('id', orderId);

      rethrow;
    }
  }

  Future<List<Map<String, dynamic>>> getOrders() async {
    final response = await _supabase
        .from('orders')
        .select('''
          id,
          pharmacy_id,
          total_price,
          order_status,
          created_at,
          pharmacies (
            name,
            address
          ),
          order_items (
            id,
            medicine_id,
            quantity,
            unit_price,
            subtotal,
            medicines (
              name,
              category,
              packaging
            )
          )
        ''')
        .order('created_at', ascending: false);

    return List<Map<String, dynamic>>.from(response);
  }
}
