class OpenFdaProduct {
  final String productNdc;
  final String genericName;
  final String brandName;
  final String manufacturer;
  final String dosageForm;
  final String productType;
  final List<String> routes;
  final List<String> activeIngredients;

  const OpenFdaProduct({
    required this.productNdc,
    required this.genericName,
    required this.brandName,
    required this.manufacturer,
    required this.dosageForm,
    required this.productType,
    required this.routes,
    required this.activeIngredients,
  });

  factory OpenFdaProduct.fromJson(Map<String, dynamic> json) {
    final ingredients = json['active_ingredients'];

    return OpenFdaProduct(
      productNdc: json['product_ndc'] ?? '',
      genericName: json['generic_name'] ?? '',
      brandName: json['brand_name'] ?? '',
      manufacturer: json['labeler_name'] ?? '',
      dosageForm: json['dosage_form'] ?? '',
      productType: json['product_type'] ?? '',
      routes: json['route'] is List
          ? List<String>.from(json['route'])
          : [],
      activeIngredients: ingredients is List
          ? ingredients
              .whereType<Map<String, dynamic>>()
              .map((item) => item['name']?.toString() ?? '')
              .where((name) => name.isNotEmpty)
              .toList()
          : [],
    );
  }
}

import 'package:flutter/foundation.dart';

class CartItem {
  final int pharmacyId;
  final String pharmacyName;
  final int medicineId;
  final String name;
  final double price;
  final int stockQuantity;
  int quantity;

  CartItem({
    required this.pharmacyId,
    required this.pharmacyName,
    required this.medicineId,
    required this.name,
    required this.price,
    required this.stockQuantity,
    this.quantity = 1,
  });

  double get subtotal => price * quantity;
}

enum CartAddResult {
  added,
  differentPharmacy,
  outOfStock,
  insufficientStock,
}

class CartProvider extends ChangeNotifier {
  final List<CartItem> _items = [];

  int? _pharmacyId;
  String? _pharmacyName;

  List<CartItem> get items => List.unmodifiable(_items);

  int? get pharmacyId => _pharmacyId;

  String? get pharmacyName => _pharmacyName;

  bool get isEmpty => _items.isEmpty;

  int get itemCount {
    return _items.fold(
      0,
      (total, item) => total + item.quantity,
    );
  }

  double get totalPrice {
    return _items.fold(
      0,
      (total, item) => total + item.subtotal,
    );
  }

  CartAddResult addItem({
    required int pharmacyId,
    required String pharmacyName,
    required int medicineId,
    required String name,
    required double price,
    required int stockQuantity,
  }) {
    if (stockQuantity <= 0) {
      return CartAddResult.outOfStock;
    }

    if (_items.isNotEmpty && _pharmacyId != pharmacyId) {
      return CartAddResult.differentPharmacy;
    }

    final existingIndex = _items.indexWhere(
      (item) => item.medicineId == medicineId,
    );

    if (existingIndex != -1) {
      final existingItem = _items[existingIndex];

      if (existingItem.quantity >= existingItem.stockQuantity) {
        return CartAddResult.insufficientStock;
      }

      existingItem.quantity++;
    } else {
      _pharmacyId = pharmacyId;
      _pharmacyName = pharmacyName;

      _items.add(
        CartItem(
          pharmacyId: pharmacyId,
          pharmacyName: pharmacyName,
          medicineId: medicineId,
          name: name,
          price: price,
          stockQuantity: stockQuantity,
        ),
      );
    }

    notifyListeners();
    return CartAddResult.added;
  }

  bool increaseQuantity(int medicineId) {
    final index = _items.indexWhere(
      (item) => item.medicineId == medicineId,
    );

    if (index == -1) return false;

    final item = _items[index];

    if (item.quantity >= item.stockQuantity) {
      return false;
    }

    item.quantity++;
    notifyListeners();
    return true;
  }

  bool decreaseQuantity(int medicineId) {
    final index = _items.indexWhere(
      (item) => item.medicineId == medicineId,
    );

    if (index == -1) return false;

    final item = _items[index];

    if (item.quantity > 1) {
      item.quantity--;
      notifyListeners();
      return true;
    }

    return false;
  }

  void removeItem(int medicineId) {
    _items.removeWhere(
      (item) => item.medicineId == medicineId,
    );

    _resetPharmacyIfEmpty();
    notifyListeners();
  }

  void clearCart() {
    _items.clear();
    _pharmacyId = null;
    _pharmacyName = null;
    notifyListeners();
  }

  void _resetPharmacyIfEmpty() {
    if (_items.isEmpty) {
      _pharmacyId = null;
      _pharmacyName = null;
    }
  }
}

import 'package:flutter/material.dart';

import '../models/openfda_product.dart';

class FavoriteProvider extends ChangeNotifier {
  final Map<String, OpenFdaProduct> _favorites = {};

  List<OpenFdaProduct> get favorites =>
      _favorites.values.toList();

  bool isFavorite(OpenFdaProduct product) {
    return _favorites.containsKey(product.productNdc);
  }

  void toggleFavorite(OpenFdaProduct product) {
    final productId = product.productNdc;

    if (productId.isEmpty) {
      return;
    }

    if (_favorites.containsKey(productId)) {
      _favorites.remove(productId);
    } else {
      _favorites[productId] = product;
    }

    notifyListeners();
  }

  void removeFavorite(OpenFdaProduct product) {
    _favorites.remove(product.productNdc);
    notifyListeners();
  }

  void clearFavorites() {
    _favorites.clear();
    notifyListeners();
  }
}

import 'package:flutter/foundation.dart';

import '../services/order_service.dart';
import 'cart_provider.dart';

class OrderProvider extends ChangeNotifier {
  final OrderService _orderService = OrderService();

  bool _isLoading = false;
  String? _errorMessage;
  Map<String, dynamic>? _lastOrder;
  List<Map<String, dynamic>> _orders = [];

  bool get isLoading => _isLoading;
  String? get errorMessage => _errorMessage;
  Map<String, dynamic>? get lastOrder => _lastOrder;
  List<Map<String, dynamic>> get orders => List.unmodifiable(_orders);

  Future<bool> createOrder(
    CartProvider cart, {
    required DateTime pickupDate,
  }) async {
    if (_isLoading) return false;

    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      _lastOrder = await _orderService.createOrder(
        cart: cart,
        pickupDate: pickupDate,
      );

      return true;
    } catch (error) {
      _errorMessage = error.toString();
      return false;
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  Future<void> loadOrders() async {
    _isLoading = true;
    _errorMessage = null;
    notifyListeners();

    try {
      _orders = await _orderService.getOrders();
    } catch (error) {
      _errorMessage = error.toString();
    } finally {
      _isLoading = false;
      notifyListeners();
    }
  }

  Future<bool> updatePickupDate({
    required int orderId,
    required DateTime pickupDate,
  }) async {
    try {
      final success = await _orderService.updatePickupDate(
        orderId: orderId,
        pickupDate: pickupDate,
      );

      if (success) await loadOrders();
      return success;
    } catch (error) {
      _errorMessage = error.toString();
      notifyListeners();
      return false;
    }
  }

  Future<bool> cancelOrder(int orderId) async {
    try {
      final success = await _orderService.cancelOrder(orderId);

      if (success) await loadOrders();
      return success;
    } catch (error) {
      _errorMessage = error.toString();
      notifyListeners();
      return false;
    }
  }
}

      _errorMessage = 'Gagal mengambil data profil: $e';
    } finally {
      _isLoading = false;
      notifyListeners();

    _isSaving = true;
    _errorMessage = null;

  String _formatDate(DateTime date) {
    const months = [
      'Januari',
      'Februari',
      'Maret',
      'April',
      'Mei',
      'Juni',
      'Juli',
      'Agustus',
      'September',
      'Oktober',
      'November',
      'Desember',
    ];

    return '${date.day} ${months[date.month - 1]} ${date.year}';
  }

  String _formatTime(DateTime date) {
    return '${date.hour.toString().padLeft(2, '0')}:'
        '${date.minute.toString().padLeft(2, '0')}';
  }

      cancelText: 'Batal',
      confirmText: 'Pilih',

        selected.month,
        selected.day,
        7,

class _FeedbackScreenState extends State<FeedbackScreen> {
  final _formKey = GlobalKey<FormState>();
  final _messageController = TextEditingController();
  final _service = FeedbackService();

  int _rating = 0;
  bool _isSubmitting = false;

  static const Color primaryColor = Color(0xFF7653B5);

  @override
  void dispose() {
    _messageController.dispose();
    super.dispose();
  }

  String? _validateMessage(String? value) {
    final text = value?.trim() ?? '';

    if (text.isEmpty) {
      return 'Feedback wajib diisi.';
    }

    if (text.length < 2) {
      return 'Feedback minimal terdiri dari 2 karakter.';
    }

    if (RegExp(r'^[0-9]+$').hasMatch(text)) {
      return 'Feedback tidak boleh hanya berisi angka.';
    }

    if (!RegExp(r'[A-Za-z]').hasMatch(text)) {
      return 'Feedback harus mengandung huruf.';
    }

    return null;
  }

  Future<void> _submitFeedback() async {
    if (!_formKey.currentState!.validate()) return;

    if (_rating == 0) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Silakan pilih rating terlebih dahulu.'),
        ),
      );
      return;
    }

    final profile = context.read<ProfileProvider>().profile;

    setState(() => _isSubmitting = true);

    try {
      await _service.submitFeedback(
        customerId: profile?.customerId,
        fullName: profile?.fullName,
        rating: _rating,
        message: _messageController.text,
      );

      if (!mounted) return;

      _messageController.clear();

      setState(() {
        _rating = 0;
        _isSubmitting = false;
      });

      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Terima kasih! Feedback berhasil dikirim.'),
          backgroundColor: Colors.green,
        ),
      );
    } catch (e) {
      if (!mounted) return;

      setState(() => _isSubmitting = false);

      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(
          content: Text('Gagal mengirim feedback: $e'),

class MedicineListScreen extends StatefulWidget {
  const MedicineListScreen({super.key});

  @override
  State<MedicineListScreen> createState() =>
      _MedicineListScreenState();
}

class _MedicineListScreenState extends State<MedicineListScreen> {
  String searchQuery = '';

  @override
  void initState() {
    super.initState();

    WidgetsBinding.instance.addPostFrameCallback((_) {
      if (!mounted) return;

      final provider = context.read<OpenFdaProvider>();

      if (provider.products.isEmpty && !provider.isLoading) {
        provider.loadProducts();
      }
    });
  }

  void _openDetail(OpenFdaProduct product) {
    Navigator.push(
      context,
      MaterialPageRoute(
        builder: (context) =>
            OpenFdaProductDetailScreen(product: product),
      ),
    );
  }

  List<OpenFdaProduct> _filterProducts(
    List<OpenFdaProduct> products,
  ) {
    final query = searchQuery.toLowerCase();

    if (query.isEmpty) {
      return products;
    }

    return products.where((product) {
      return product.brandName.toLowerCase().contains(query) ||
          product.genericName.toLowerCase().contains(query) ||
          product.manufacturer.toLowerCase().contains(query);
    }).toList();
  }

  Widget _buildProductCard(OpenFdaProduct product) {
    final displayName = product.brandName.isNotEmpty
        ? product.brandName
        : product.genericName;

    return Card(
      elevation: 1,
      color: Colors.white,
      shape: RoundedRectangleBorder(
        borderRadius: BorderRadius.circular(14),
      ),
      clipBehavior: Clip.antiAlias,
      child: InkWell(
        onTap: () => _openDetail(product),
        child: Padding(
          padding: const EdgeInsets.all(16),
          child: Row(
            crossAxisAlignment: CrossAxisAlignment.start,
            children: [
              Container(
                width: 48,
                height: 48,
                decoration: BoxDecoration(
                  color: AppTheme.blueAccent,
                  borderRadius: BorderRadius.circular(12),
                ),
                child: const Icon(
                  Icons.medication_outlined,
                  color: AppTheme.primaryColor,
                ),
              ),
              const SizedBox(width: 14),
              Expanded(
                child: Column(
                  crossAxisAlignment: CrossAxisAlignment.start,
                  children: [
                    Text(
                      displayName.isNotEmpty
                          ? displayName
                          : 'Nama produk tidak tersedia',
                      style: const TextStyle(
                        fontWeight: FontWeight.bold,
                        fontSize: 16,
                      ),
                    ),
                    if (product.genericName.isNotEmpty) ...[
                      const SizedBox(height: 5),
                      Text(
                        'Nama generik: ${product.genericName}',
                        style: const TextStyle(fontSize: 13),
                      ),
                    ],
                    if (product.manufacturer.isNotEmpty) ...[
                      const SizedBox(height: 5),
                      Text(
                        'Produsen: ${product.manufacturer}',
                        style: TextStyle(
                          fontSize: 12,
                          color: Colors.grey.shade700,
                        ),
                      ),
                    ],
                    if (product.dosageForm.isNotEmpty) ...[
                      const SizedBox(height: 5),
                      Text(
                        'Bentuk sediaan: ${product.dosageForm}',
                        style: const TextStyle(fontSize: 12),
                      ),
                    ],
                    const SizedBox(height: 8),
                    Row(
                      children: [
                        Text(
                          'Lihat detail produk',
                          style: TextStyle(
                            fontSize: 12,
                            color: AppTheme.primaryColor,
                            fontWeight: FontWeight.w600,
                          ),
                        ),
                        const SizedBox(width: 4),
                        const Icon(
                          Icons.arrow_forward,
                          size: 14,
                          color: AppTheme.primaryColor,
                        ),
                      ],
                    ),
                  ],
                ),
              ),
            ],
          ),
        ),
      ),
    );
  }

  Widget _buildContent(OpenFdaProvider provider) {
    if (provider.isLoading) {
      return const Center(
        child: CircularProgressIndicator(),
      );
    }

    if (provider.errorMessage != null) {
      return Center(
        child: Column(
          mainAxisSize: MainAxisSize.min,
          children: [
            const Icon(
              Icons.cloud_off_outlined,
              size: 48,
              color: Colors.grey,
            ),
            const SizedBox(height: 12),
            Text(
              provider.errorMessage!,
              textAlign: TextAlign.center,
            ),
            const SizedBox(height: 12),
            ElevatedButton.icon(
              onPressed: provider.loadProducts,
              icon: const Icon(Icons.refresh),
              label: const Text('Coba Lagi'),
            ),
          ],
        ),
      );
    }

    if (provider.products.isEmpty) {
      return const Center(
        child: Text('Belum ada data produk.'),
      );
    }

    final filteredProducts = _filterProducts(provider.products);

    if (filteredProducts.isEmpty) {
      return const Center(
        child: Text('Produk tidak ditemukan.'),
      );
    }

    return LayoutBuilder(
      builder: (context, constraints) {
        if (constraints.maxWidth >= 700) {
          return GridView.builder(
            itemCount: filteredProducts.length,
            gridDelegate:
                const SliverGridDelegateWithMaxCrossAxisExtent(
              maxCrossAxisExtent: 450,
              mainAxisExtent: 170,
              crossAxisSpacing: 12,
              mainAxisSpacing: 12,
            ),
            itemBuilder: (context, index) {
              return _buildProductCard(
                filteredProducts[index],
              );
            },
          );
        }

        return ListView.separated(
          itemCount: filteredProducts.length,
          separatorBuilder: (context, index) =>
              const SizedBox(height: 12),
          itemBuilder: (context, index) {
            return _buildProductCard(
              filteredProducts[index],
            );
          },
        );
      },
    );
  }

  Widget build(BuildContext context) {
    return Consumer<OpenFdaProvider>(
      builder: (context, provider, child) {
        final filteredProducts =
            _filterProducts(provider.products);

        return Scaffold(

                TextField(
                  onChanged: (value) {

                    hintText: 'Cari nama atau produsen produk...',
                    prefixIcon: const Icon(Icons.search),
                    filled: true,
                    fillColor: AppTheme.blueAccent,
                    border: OutlineInputBorder(
                      borderRadius: BorderRadius.circular(14),
                      borderSide: BorderSide.none,
                    ),
                  ),
                ),
                const SizedBox(height: 16),
                Text(
                  'Data Produk OpenFDA (${filteredProducts.length})',
                  style: const TextStyle(
                    fontSize: 18,
                    fontWeight: FontWeight.bold,
                  ),
                ),
                const SizedBox(height: 12),
                Expanded(
                  child: _buildContent(provider),
                ),
                const SizedBox(height: 8),
                Text(
                  'Sumber informasi: OpenFDA. '
                  'Data ini merupakan informasi produk, '
                  'bukan rekomendasi penggunaan obat.',
                  style: TextStyle(
                    fontSize: 11,
                    color: Colors.grey.shade600,
                  ),
                ),
              ],
            ),
          ),
        );
      },

  String _formatPrice(dynamic price) {
    final value = double.tryParse(price.toString()) ?? 0;

    return 'Rp ${value.toStringAsFixed(0).replaceAllMapped(
      RegExp(r'(\d)(?=(\d{3})+(?!\d))'),
      (match) => '${match[1]}.',
    )}';
  }

  String _formatDate(dynamic date) {
    if (date == null) return 'Belum ditentukan';

    final parsed = DateTime.tryParse(date.toString());
    if (parsed == null) return '-';

    final d = parsed.toLocal();

    const months = [
      'Januari', 'Februari', 'Maret', 'April',
      'Mei', 'Juni', 'Juli', 'Agustus',
      'September', 'Oktober', 'November', 'Desember',
    ];

    return '${d.day} ${months[d.month - 1]} ${d.year}, '
        '${d.hour.toString().padLeft(2, '0')}:'
        '${d.minute.toString().padLeft(2, '0')}';
  }

  Map<String, dynamic> _asMap(dynamic value) {
    if (value is Map) {
      return Map<String, dynamic>.from(value);
    }
    return {};
  }

  List<Map<String, dynamic>> _asMapList(dynamic value) {
    if (value is List) {
      return value
          .whereType<Map>()
          .map((item) => Map<String, dynamic>.from(item))
          .toList();
    }
    return [];
  }

  String _statusLabel(String status) {
    switch (status.toLowerCase()) {
      case 'pending':
        return 'Menunggu';
      case 'processing':
        return 'Diproses';
      case 'completed':
        return 'Selesai';
      case 'cancelled':
        return 'Dibatalkan';
      default:
        return status;
    }
  }

  Color _statusColor(String status) {
    switch (status.toLowerCase()) {
      case 'completed':
        return Colors.green;
      case 'cancelled':
        return Colors.red;
      case 'processing':
        return Colors.blue;
      case 'pending':
        return Colors.orange;
      default:
        return Colors.grey;
    }
  }

  Future<void> _changePickupDate(
    Map<String, dynamic> order,
  ) async {
    final orderId = int.tryParse(order['id'].toString());
    if (orderId == null) return;

    final currentPickup = order['pickup_at'] == null
        ? null
        : DateTime.tryParse(order['pickup_at'].toString())?.toLocal();

    final now = DateTime.now();

    final selected = await showDatePicker(
      context: context,
      initialDate: currentPickup != null &&
              currentPickup.isAfter(now)
          ? currentPickup
          : now,
      firstDate: DateTime(now.year, now.month, now.day),
      lastDate: DateTime(now.year + 1, 12, 31),
      helpText: 'Pilih tanggal pengambilan',
    );

    if (selected == null || !mounted) return;

    final selectedTime = await showTimePicker(
      context: context,
      initialTime: currentPickup != null
          ? TimeOfDay.fromDateTime(currentPickup)
          : TimeOfDay.now(),
      helpText: 'Pilih jam pengambilan',
    );

    if (selectedTime == null || !mounted) return;

    final pickupDate = DateTime(
      selected.year,
      selected.month,
      selected.day,
      selectedTime.hour,
      selectedTime.minute,
    );

    if (pickupDate.isBefore(DateTime.now())) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(
          content: Text('Tanggal dan jam harus di masa depan.'),
        ),
      );
      return;
    }

    final success = await context.read<OrderProvider>()
        .updatePickupDate(
          orderId: orderId,
          pickupDate: pickupDate,
        );

    if (!mounted) return;

    ScaffoldMessenger.of(context).showSnackBar(
      SnackBar(
        content: Text(
          success
              ? 'Tanggal pengambilan berhasil diperbarui.'
              : 'Gagal mengubah tanggal. Pastikan pesanan masih menunggu.',
        ),
      ),
    );
  }

class OpenFdaInventoryService {
  final SupabaseClient _supabase = Supabase.instance.client;

  Future<bool> saveProduct(OpenFdaProduct product) async {

      throw Exception('Kode produk NDC tidak tersedia.');
    }

    final existing = await _supabase
        .from('medicines')
        .select('id')
        .eq('product_ndc', product.productNdc)
        .maybeSingle();

    if (existing != null) {
      return false;
    }

    final description = [
      'Nama generik: ${product.genericName}',
      'Produsen: ${product.manufacturer}',
      'Bahan aktif: ${product.activeIngredients.join(', ')}',
    ].join('\n');

    await _supabase.from('medicines').insert({
      'product_ndc': product.productNdc,
      'name': product.brandName.isNotEmpty
          ? product.brandName
          : product.genericName,
      'category': product.productType,
      'description': description,
      'packaging': product.dosageForm,
      'stock': 0,
    });

    return true;
  }
}

import 'dart:convert';

import 'package:http/http.dart' as http;

import '../models/openfda_product.dart';

class OpenFdaService {
  static const String _baseUrl =
      'https://api.fda.gov/drug/ndc.json';

  Future<List<OpenFdaProduct>> getProducts() async {
    final uri = Uri.parse(_baseUrl).replace(
      queryParameters: {
        'search': 'product_type:"HUMAN OTC DRUG"',
        'limit': '20',
      },
    );

    final response = await http.get(uri);

    if (response.statusCode == 200) {
      final Map<String, dynamic> data =
          jsonDecode(response.body) as Map<String, dynamic>;

      final List<dynamic> results =
          data['results'] as List<dynamic>;

      return results
          .map(
            (item) => OpenFdaProduct.fromJson(
              item as Map<String, dynamic>,
            ),
          )
          .toList();
    }

    throw Exception(
      'Gagal mengambil data OpenFDA. '
      'Status code: ${response.statusCode}',
    );
  }
}

import 'package:supabase_flutter/supabase_flutter.dart';

import '../providers/cart_provider.dart';

class OrderService {
  final SupabaseClient _supabase = Supabase.instance.client;

  Future<Map<String, dynamic>> createOrder({
    required CartProvider cart,
    required DateTime pickupDate,
  }) async {
    if (cart.isEmpty || cart.pharmacyId == null) {
      throw Exception('Keranjang masih kosong.');
    }

    final order = await _supabase
        .from('orders')
        .insert({
          'pharmacy_id': cart.pharmacyId,
          'total_price': cart.totalPrice,
          'order_status': 'pending',
          'pickup_at': pickupDate.toUtc().toIso8601String(),
        })
        .select()
        .single();

    final orderId = order['id'];

    try {
      final orderItems = cart.items.map((item) {
        return {
          'order_id': orderId,
          'medicine_id': item.medicineId,
          'quantity': item.quantity,
          'unit_price': item.price,
          'subtotal': item.subtotal,
        };
      }).toList();

      await _supabase.from('order_items').insert(orderItems);

      return order;
    } catch (error) {
      await _supabase
          .from('orders')
          .delete()
          .eq('id', orderId);

      rethrow;
    }
  }

  Future<List<Map<String, dynamic>>> getOrders() async {
    final response = await _supabase.from('orders').select('''
      id,
      pharmacy_id,
      total_price,
      order_status,
      created_at,
      pickup_at,
      pharmacies (
        name,
        address
      ),
      order_items (
        id,
        medicine_id,
        quantity,
        unit_price,
        subtotal,
        medicines (
          name,
          category,
          packaging
        )
      )
    ''').order('created_at', ascending: false);

    return List<Map<String, dynamic>>.from(response);
  }

  Future<bool> updatePickupDate({
    required int orderId,
    required DateTime pickupDate,
  }) async {
    final result = await _supabase
        .from('orders')
        .update({
          'pickup_at': pickupDate.toUtc().toIso8601String(),
        })
        .eq('id', orderId)
        .eq('order_status', 'pending')
        .select('id')
        .maybeSingle();

    return result != null;

import 'dart:io';

import 'package:supabase_flutter/supabase_flutter.dart';

class ProfileService {
  final SupabaseClient _client = Supabase.instance.client;

  static const String tableName = 'profiles';
  static const String bucketName = 'profile-photos';
  static const int profileId = 1;

  // Mengambil data profil dummy dengan ID 1.
  Future<Map<String, dynamic>> getProfile() async {
    final response = await _client
        .from(tableName)
        .select()
        .eq('id', profileId)
        .single();

    return response;
  }

  // Memperbarui informasi profil.
  Future<void> updateProfile(Map<String, dynamic> data) async {
    await _client.from(tableName).update({
      ...data,
      'updated_at': DateTime.now().toUtc().toIso8601String(),
    }).eq('id', profileId);
  }

  // Mengunggah foto profil ke Supabase Storage.
  Future<String> uploadProfilePhoto(File imageFile) async {
    final extension = imageFile.path.split('.').last.toLowerCase();

    final contentType = switch (extension) {
      'png' => 'image/png',
      'webp' => 'image/webp',
      'jpg' || 'jpeg' => 'image/jpeg',
      'jfif' => 'image/jpeg',
      _ => 'application/octet-stream',
    };

    final fileName =
        'profile_$profileId.${DateTime.now().millisecondsSinceEpoch}.$extension';

    await _client.storage.from(bucketName).upload(
      fileName,
      imageFile,
      fileOptions: FileOptions(
        contentType: contentType,
        upsert: true,
      ),
    );

    return _client.storage.from(bucketName).getPublicUrl(fileName);
  }
}
